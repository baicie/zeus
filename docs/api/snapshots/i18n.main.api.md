/**
 * Locale identity and negotiation.
 *
 * The supported set is deliberately closed: a locale this package does not have
 * a catalog for must resolve to one it does, rather than half-render. Adding a
 * language later means adding a catalog and an entry here — never widening the
 * type to `string`, which would let a typo become a silent fallback.
 */
/** Locales with a bundled catalog. The first is the default fallback. */
declare const SUPPORTED_LOCALES: readonly ['en', 'zh-CN']
type Locale = (typeof SUPPORTED_LOCALES)[number]
declare const DEFAULT_LOCALE: Locale
/**
 * Resolves any BCP 47 tag to a supported locale, or `undefined`.
 *
 * Returns `undefined` rather than falling back so the caller can distinguish
 * "the user asked for this and we chose it" from "we had no catalog and fell
 * back". Both end at a supported locale, but only the first should persist.
 */
declare function resolveLocale(
  input: string | undefined | null,
): Locale | undefined
/** Resolves, or returns the default when nothing matched. */
declare function normalizeLocale(input: string | undefined | null): Locale
/**
 * Picks the first supported locale from a preference list.
 *
 * Order is respected exactly: `['ja', 'zh-CN']` yields `zh-CN`, and
 * `['zh-TW', 'en']` yields `en` — a Traditional request falls past Simplified
 * to the next candidate rather than being coerced into it.
 */
declare function negotiateLocale(
  candidates: readonly string[] | undefined | null,
): Locale | undefined
declare function isSupportedLocale(value: unknown): value is Locale
/**
 * Reflects the active locale on `<html lang>`.
 *
 * Accessibility tools and the browser's own hyphenation read this attribute;
 * leaving it at the document's original value while the UI renders Chinese
 * makes a screen reader announce Chinese text with an English voice. No-ops
 * outside a document (SSR, tests).
 */
declare function applyDocumentLocale(locale: Locale): void

/**
 * Message catalogs and the typed-key contract.
 *
 * An application declares its English catalog as the source of truth and
 * derives its key type from it. Every other locale is then checked against
 * that key set by the compiler (`satisfies`), and again at runtime by
 * `checkCatalog` — `satisfies` catches a missing key in a literal, but a
 * generated or merged catalog can still be short, and a user who switches to
 * that language must not see raw keys.
 */
/** Values a message may interpolate. */
type MessageParam = string | number | Date
type MessageParams = Readonly<Record<string, MessageParam>>
/** A catalog: flat, dotted keys mapping to ICU message text. */
type MessageCatalog<K extends string = string> = Readonly<Record<K, string>>
/** Keys of a catalog object. */
type MessageKeyOf<C> = Extract<keyof C, string>
/**
 * A message held as data rather than as rendered text.
 *
 * State that outlives a render — a toast, a failed request, a status line —
 * must keep the key and its parameters, not the string. Storing the string
 * means the message stays in the language it was produced in after the user
 * switches, which is the mixed-language UI this package exists to prevent.
 *
 * `text` is for content that is not translatable by definition: a server's raw
 * error message with no code, a file path, a compiler diagnostic.
 */
type MessageDescriptor<K extends string = string> =
  | {
      readonly key: K
      readonly params?: MessageParams
    }
  | {
      readonly text: string
    }
declare function isKeyDescriptor<K extends string>(
  descriptor: MessageDescriptor<K>,
): descriptor is {
  readonly key: K
  readonly params?: MessageParams
}
interface CatalogIssue {
  readonly kind:
    | 'missing-key'
    | 'extra-key'
    | 'duplicate-key'
    | 'empty-message'
    | 'same-as-reference'
  readonly locale: string
  readonly key: string
  readonly referenceLocale?: string
}
interface CatalogCheckOptions {
  /** Locale the catalog is checked against; defaults to `en`. */
  readonly referenceLocale?: string
  /**
   * Keys whose translation legitimately equals the reference text.
   *
   * Without this, a translator who leaves "OK" as "OK" is reported as a missed
   * translation. Every entry is a claim that the two languages agree here.
   */
  readonly allowIdenticalKeys?: readonly string[]
}
/**
 * Compares a catalog against the reference key set.
 *
 * Returns issues rather than throwing so a CI check can report all of them;
 * `assertCatalog` is the fail-fast variant for call sites that want it.
 */
declare function checkCatalog<K extends string>(
  reference: MessageCatalog<K>,
  locale: string,
  candidate: MessageCatalog<string>,
  options?: CatalogCheckOptions,
): readonly CatalogIssue[]
/** Throws listing every issue; used by tests and by `createI18n` in dev. */
declare function assertCatalog<K extends string>(
  reference: MessageCatalog<K>,
  locale: string,
  candidate: MessageCatalog<string>,
  options?: CatalogCheckOptions,
): void
/**
 * Fills `{name}`-style placeholders without ICU.
 *
 * Only used as the last-resort fallback when a message fails to compile at
 * runtime and the raw template must still be shown. Normal rendering goes
 * through the ICU formatter, so plural rules and number formatting apply.
 */
declare function interpolate(
  template: string,
  params: MessageParams | undefined,
): string

/**
 * Message formatting and Intl formatter caching.
 *
 * A compiled ICU message is an object graph; parsing one per render would put
 * the parse cost inside every reactive update, and an SSE stream re-renders the
 * same message many times per second. Caches are keyed by locale and template
 * text, and bounded so a session with many distinct messages cannot grow
 * without limit.
 *
 * Formatters are locale-scoped by construction: a cache entry never outlives
 * its locale, so switching languages cannot reuse an English formatter.
 */

/**
 * Compiles `template` for `locale`, tolerating a malformed message.
 *
 * An invalid ICU string is a content bug, and CI rejects it (see
 * `checkMessageSyntax`), but a bad catalog shipped to production must not take
 * the page down. The fallback interpolates the raw template so the visible
 * result is the author's text with values filled in, not a crash and not a raw
 * key.
 */
declare function formatIcuMessage(
  locale: Locale,
  template: string,
  params: MessageParams | undefined,
  onInvalid?: (locale: Locale, template: string, error: Error) => void,
): string

  locale: Locale,
  value: number,
  options?: Intl.NumberFormatOptions,
): string

  locale: Locale,
  value: Date | number,
  options?: Intl.DateTimeFormatOptions,
): string
declare function formatRelative(
  locale: Locale,
  value: number,
  unit: Intl.RelativeTimeFormatUnit,
  options?: Intl.RelativeTimeFormatOptions,
): string
/**
 * Parses a message without formatting it, for CI syntax checks.
 *
 * Returns the error instead of throwing so a checker can report every bad
 * message in one run rather than stopping at the first.
 */
declare function checkMessageSyntax(
  locale: Locale,
  template: string,
): Error | undefined
/**
 * Placeholder names a message requires, from its compiled AST.
 *
 * CI compares this set across locales: a Chinese message that drops `{count}`
 * renders "位用户" with no number, and no amount of type checking on the key
 * alone would catch it.
 */
declare function messageArgumentNames(
  locale: Locale,
  template: string,
): readonly string[]
/** Drops every compiled message and formatter. Used between tests. */
declare function clearFormatterCaches(): void

/**
 * The i18n runtime.
 *
 * One `createI18n()` call yields an isolated instance: its own locale signal,
 * its own catalogs, its own formatters. Nothing is module-global, so two
 * instances can render different languages in the same page (a settings
 * preview) and a server can hold one per request.
 *
 * Switching locale is atomic. A new catalog is loaded and validated on its own,
 * and only then is the reactive locale committed — so no render ever observes
 * a half-applied language, and a failed load leaves the previous language
 * intact rather than showing keys.
 */

/** Lifecycle of the active catalog. */
type I18nStatus = 'ready' | 'loading' | 'error'
interface I18nEvents {
  /** A key missing from the active catalog and its fallback. */
  onMissingKey?: (locale: Locale, key: string) => void
  /** A catalog failed to load; the previous locale stays active. */
  onLoadError?: (locale: Locale, error: Error) => void
  /** A message failed to compile or format; raw text was shown instead. */
  onInvalidMessage?: (locale: Locale, template: string, error: Error) => void
}
interface CreateI18nOptions<K extends string> extends I18nEvents {
  /**
   * Catalogs available synchronously.
   *
   * The reference locale must be here regardless of how the app loads the
   * others: it is the fallback every render depends on, and a fallback that
   * arrives asynchronously would let the first paint show raw keys.
   */
  readonly messages: Partial<Record<Locale, MessageCatalog<K>>> & {
    readonly en: MessageCatalog<K>
  }
  /** Initial locale; defaults to `en`. */
  readonly initialLocale?: Locale
  /** Locale used when a key is missing or a load fails. Defaults to `en`. */
  readonly fallbackLocale?: Locale
  /**
   * Loads a catalog that is not in `messages`.
   *
   * Called at most once per locale unless a previous attempt failed — a
   * rejected load is not cached, so a retry is possible.
   */
  readonly loadCatalog?: (locale: Locale) => Promise<MessageCatalog<K>>
  /** Mirrors the active locale onto `<html lang>`. Defaults to `true`. */
  readonly syncDocumentLang?: boolean
  /**
   * Rejects a catalog whose key set differs from the reference.
   *
   * On by default in development. In production a malformed catalog still
   * renders — the missing messages fall back to the reference language — so a
   * content mistake degrades instead of blanking the page.
   */
  readonly strictCatalogs?: boolean
}
interface I18n<K extends string> {
  /** The active locale. Reading it inside a render subscribes to changes. */
  readonly locale: () => Locale
  /** `ready` unless a `setLocale` is in flight or the last one failed. */
  readonly status: () => I18nStatus
  /** The error from the last failed `setLocale`, if any. */
  readonly error: () => Error | undefined
  /**
   * Switches the active locale.
   *
   * Resolves once the new catalog is active. Rejects if it could not be
   * loaded, leaving the previous locale in place. Concurrent calls resolve to
   * the last request made, not the last to finish.
   */
  setLocale(locale: Locale): Promise<void>
  /** Pre-loads a catalog without switching to it. */
  preload(locale: Locale): Promise<void>
  t(key: K, params?: MessageParams): string
  /** Captures a message for later rendering (toasts, status, errors). */
  message(key: K, params?: MessageParams): MessageDescriptor<K>
  /** Wraps untranslatable text (server messages, paths, diagnostics). */
  raw(text: string): MessageDescriptor
  /** Renders a descriptor in the current locale. */
  resolve(descriptor: MessageDescriptor<K>): string
  formatNumber(value: number, options?: Intl.NumberFormatOptions): string
  formatDate(value: Date | number, options?: Intl.DateTimeFormatOptions): string
  formatRelative(
    value: number,
    unit: Intl.RelativeTimeFormatUnit,
    options?: Intl.RelativeTimeFormatOptions,
  ): string
  /** Releases the instance's signal; it must not be used afterwards. */
  dispose(): void
}
declare function createI18n<K extends string>(
